\begin{frame}{Compression par grammaires}

  \on[width=1.1\textwidth, top=-3mm]{
    \Probleme{SmallestGrammar}{
      \begin{itemize}
      \item Un mot $w \in \Sigma^\star$
      \item Un entier $k$
      \end{itemize}
    }{
      Existe-t-il une grammaire algébrique  $G = \langle \Sigma, \Gamma, S, \rightarrow \rangle$ tq :
      \begin{itemize}
      \item \structure{$\mathcal{L}(G) = \{ w \}$}, et
      \item la taille de $G$ est au plus $k$ : \structure{$\displaystyle\sum_{A \rightarrow \beta} |\beta| \le k$} ?
      \end{itemize}
    }
  }

  \onBlock[y=2mm, anchor=north]{Théorème (admis)}{
    \textsc{SmallestGrammar} est \textsc{np}-complet.
  }

  \onExampleBlock[bottom=3mm]{Exemple pour $w = \texttt{ababababab}$}{
    Une grammaire minimale pour $w$ est $G_w$ :
    
    \quad\quad$G : \left\{\begin{array}{@{~}r@{~}c@{~}l@{~}}
    S & \rightarrow & ABA \\
    A & \rightarrow & BB \\
    B & \rightarrow & ab
    \end{array}\right.$ \quad(taille $7$)
  }

  \on[x=30mm, y=-20mm]{
    \begin{tikzpicture}[tree, x=5mm, y=6.5mm, tree node internal/.append style={alert,}, tree node leave/.append style={example,},]
      \tree{$S$}{

          }
        }
      }
    \end{tikzpicture}
  }
