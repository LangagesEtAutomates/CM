\begin{frame}{Séquençage du génome}

  \on[text, top=-4mm]{
    \Probleme{ShortestSuperstring}{
      \begin{itemize}
      \item Un langage fini $S = \{s_1, ..., s_k\} \subset \Sigma^\star$ 
      \item Un entier $\ell$
      \end{itemize}
    }{
      Existe-t-il un mot $w \in \Sigma^\star$ tel que :
      \begin{itemize}
      \item $|w| \le \ell$, et
      \item pour tout $s_i\in S$, $s_i$ est un facteur de $w$ ?
      \end{itemize}
    }
  }
  
  \onBlock[y=6mm, anchor=north]{Théorème (admis)}{
    \textsc{ShortestSuperstring} est \textsc{np}-complet.
  }

  \onExampleBlock[y=-7mm, anchor=north]{Exemple : $S = \{ \textsc{tagcg}, \textsc{gcgat}, \textsc{acagt}, \textsc{agtag} \}, \ell = 11$}{
