\begin{frame}{Théorème de Cook-Levin}

  \onBlock[top=-5mm]{Théorème -- Cook-Levin (1971)}{
    Le problème \textsc{sat} est \textsc{np}-complet
  }

  \onBlock<2->[left=.85\textwidth]{Idées de la démonstration (admise)}{

    \structure{$\textsc{sat}\in \textsc{np}$ :} Un certificat est une valuation

    \vspace{1mm}
    \structure{$\textsc{sat}\in \textsc{np-difficile}$ :} On montre $\textsc{NPUniversal} \leq_p \textsc{sat}$

    Variables principales de la formule construite :

    \begin{description}[$r_{a, t, i}$ :]
    \item[$r_{a, t, i}$ :] pour $a \in \Gamma$, $t\le k$ et $i\in \llbracket -k, k+|u| \rrbracket$
      \begin{itemize}
      \item $a$ est écrit dans la case $i$ après $t$ actions
      \end{itemize}
    \item[$s_{q, t}$ :] pour $q \in Q$, $t\le k$
      \begin{itemize}
      \item la machine est dans l'état $q$ après $t$ actions
      \end{itemize}
    \item[$h_{t, i}$ :] pour $t\le k$ et $i\in \llbracket -k, k+|u| \rrbracket$
      \begin{itemize}
      \item la tête de lecture est sous la case $i$ après $t$ actions
      \end{itemize}
    \end{description}
  }
