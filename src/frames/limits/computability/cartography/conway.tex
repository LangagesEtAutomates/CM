  \on[text, top=-4mm]{
    \Probleme{game of life}{
      Un ensemble fini de \structure{cellules vivantes} de $\mathbb{Z}^2$
    }{
      Existera-t-il toujours \structure{au moins une cellule vivante} ?
      \begin{itemize}
      \item Une cellule est vivante à l'étape suivante si :
        \begin{itemize}
        \item elle a 3 voisines vivantes, ou 
        \item elle a 2 voisines vivantes et elle est vivante
        \end{itemize}
      \end{itemize}
    }
  }
  
  \onBlock[y=6mm, anchor=north]{Théorème (admis)}{
    \textsc{game of life} est indécidable.
  }

  \onExampleBlock[y=-7mm, anchor=north]{Exemple}{
