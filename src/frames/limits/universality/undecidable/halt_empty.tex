\begin{frame}{Problème de l'arrêt sur chaîne vide}

  \on[text, top=-3mm]{
    Soit $\langle \mathcal{L}, \llbracket \cdot \rrbracket \rangle$ un encodage effectif.

    \Probleme{$\textsc{Halt-}\varepsilon_{\langle \mathcal{L}, \llbracket \cdot \rrbracket \rangle}$}{
      un \structure{programme} $P \in \mathcal{L}$
    }{
      Est-ce que l'exécution de $P$ sur $\varepsilon$ termine ? 
    }
  }
  
  \onBlock[y=5mm]{Théorème -- $\textsc{Halt-}\varepsilon_{\langle \mathcal{L}, \llbracket \cdot \rrbracket \rangle}$ est indécidable}{
    On le montre par réduction depuis $\textsc{Halt}_{\langle \mathcal{L}, \llbracket \cdot \rrbracket \rangle}$. 
  }
  \on[text, y=-23mm]<2>{
    On pose 
    $
    f : \left\{\begin{array}{@{~}r@{~~}c@{~~}l@{~}}
    \mathcal{I}\left(\textsc{Halt}\right) & \rightarrow & \mathcal{I}\left(\textsc{Halt-}\varepsilon\right)\\
    \langle P, u \rangle &\mapsto& P', \text{ où : }~ \begin{algorithm}[H]
      \Structure{\lFun{$P'(x) : \mathbb{B}$}{
        \Return $P(u)$;
      }}
    \end{algorithm}
    \end{array}\right.
    $

    \vspace{1mm}
    \begin{itemize}
    \item $f$ est calculable
      \begin{itemize}
      \item \structure{Remarque.} $f : (('a \rightarrow \texttt{bool}) \times {'a}) \rightarrow 'a \rightarrow \texttt{bool}$ est curryfiée
      \end{itemize}
    \item $P'$ termine sur $\varepsilon$ si, et seulement si, $P$ termine sur $u$
      \begin{itemize}
      \item Donc $\langle P,u \rangle \in \textsc{pos}(\textsc{halt}) \Leftrightarrow f(\langle P,u \rangle) \in \textsc{pos}(\textsc{halt-}\varepsilon)$
      \end{itemize}
    \item $f$ est une réduction de $\textsc{halt}$ vers $\textsc{halt-}\varepsilon$, et $\textsc{halt}$ est indécidable
      \begin{itemize}
      \item Donc $\textsc{halt-}\varepsilon$ est indécidable
      \end{itemize}
    \end{itemize}
  }
\end{frame}
