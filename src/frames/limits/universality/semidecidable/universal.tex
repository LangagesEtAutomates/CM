\begin{frame}{Machine de Turing universelle}
  
  \onBlock[top=-5mm]{Définition -- Machine de Turing universelle}{

    \vspace{-1mm}
    Une machine de Turing $M_{univ}$ est dite \structure{universelle} sur un alphabet $\Sigma$ si :

    \begin{itemize}
    \item $M_{univ}$ prend en entrée : 
      \begin{itemize}
      \item un \structure{encodage} $\mathit{encode}(M)$ d'une MTD $M$,
      \item un \structure{mot} $u \in \Sigma^\star$,
      \end{itemize}
    \item et $M_{univ}$ \structure{simule} l'exécution de $M$ sur $u$ :
      \begin{itemize}
      \item \structure{$M_{univ}$ termine sur $\langle \mathit{encode}(M), u \rangle$} si, et seulement si, \structure{$M$ termine sur $u$};
      \item \structure{$M_{univ}$ accepte $\langle \mathit{encode}(M), u \rangle$} si, et seulement si, \structure{$M$ accepte $u$}. 
      \end{itemize}
    \item C'est à dire, $M_{univ}$ semi-décide \alert{$L_{univ} \eqdef \{\langle \mathit{encode}(M), u \rangle \mid M \text{ accepte } u \}$}
    \end{itemize}
  }

  \onExampleBlock<2->[y=-2mm, anchor=north]{Exemple d'encodage :}{

    \vspace{-1mm}
    On encode une suite de transitions :
    \begin{itemize}
    \item Encodage des états en taille fixe
    \item $\mathit{encode}\left(\example{q} \structure{\xrightarrow{\smTMtrans{\alert{a}}{\alert{b}}{d}}} \example{q'} \right) = \alert{a} \cdot \example{q} \cdot \structure{\rightarrow} \cdot \alert{a'} \cdot \example{q'^{\textsc{r}}} \cdot \structure{d}$
    \end{itemize}
  }
